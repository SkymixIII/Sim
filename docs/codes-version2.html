Codes tally version 2

Les deux codes de la version 2 : l'émetteur WT32-ETH01 avec interrupteurs de M/E, et le tally ESP32-C3 avec afficheur et boutons + et −. Le même code tally sert pour tous les boîtiers : l'entrée se règle ensuite aux boutons.

Émetteur · emetteur_v2.ino

Carte : ESP32 Dev Module. Aucune bibliothèque. Réglages en haut du fichier : IP_ATEM, IP_EMETTEUR. Les M/E se choisissent avec les interrupteurs sur IO4, IO14, IO32 et IO33. Téléversement par l'adaptateur USB-série sur J2, avec IO0 relié à GND.

// ============================================================================
//  VERSION 2 · Émetteur tally ATEM -> ESP-NOW  ·  WT32-ETH01 + interrupteurs de M/E
// ----------------------------------------------------------------------------
//  Le WT32-ETH01 est branché à l'ATEM par son RJ45. Il parle lui-même
//  le protocole ATEM (UDP 9910) et calcule ce qui est à l'antenne sur chaque M/E :
//    - source de programme et de preview ;
//    - clés en incrustation (DVE, luma, chroma, motif) : leur source de remplissage ;
//    - SuperSource : sources des boîtes actives et le fond ;
//    - multiview envoyé comme source : les sources de ses fenêtres ;
//    - sortie d'une M/E utilisée dans une autre, DSK (sur la M/E 1), transitions.
//  Le tally est diffusé en ESP-NOW (trame v2, une case par entrée SDI),
//  10 fois par seconde et tout de suite à chaque changement.
//
//  Interrupteurs de choix des M/E (facultatifs) :
//    ME1 -> IO4   ME2 -> IO14   ME3 -> IO32   ME4 -> IO33   (l'autre côté à GND)
//    ON = M/E suivie. Aucun interrupteur câblé, ou tous sur OFF = toutes les M/E.
//    Une entrée est ROUGE si elle est au programme de l'une des M/E suivies,
//    sinon VERTE si elle est en preview sur l'une d'elles.
//  (Ne pas utiliser IO12 : s'il est tiré à 3,3 V au démarrage, la carte ne démarre pas.)
//
//  Alimentation : 5V et GND du WT32-ETH01 (ou 3V3 et GND).
//  Téléversement : adaptateur USB-série 3,3 V : TX -> RX0, RX -> TX0, GND -> GND,
//  et IO0 relié à GND pendant la mise sous tension (à retirer ensuite).
//  LED sur IO2 : allumée = pas connecté à l'ATEM.
//
//  Aucune bibliothèque externe.  Carte : « ESP32 Dev Module » (ou « WT32-ETH01 Ethernet Module »).
//  Moniteur série : 115200 bauds.
// ============================================================================
#include <ETH.h>
#include <WiFi.h>
#include <WiFiUdp.h>
#include <esp_now.h>
#include <esp_wifi.h>

// ---------- Réglages ----------
static const uint8_t  IP_ATEM[4] = {192, 168, 2, 20};
static const uint8_t  IP_EMETTEUR[4] = {192, 168, 2, 52};  // adresse du WT32 : doit être libre sur le réseau
static const uint8_t  MASQUE[4] = {255, 255, 255, 0};
static const uint8_t  NB_ENTREES = 20;      // entrées SDI transmises (40 au plus)
static const uint8_t  CANAL_ESPNOW = 1;
static const uint16_t PORT_ATEM = 9910;
static const uint16_t PORT_LOCAL = 50910;
static const uint32_t PERIODE_RADIO_MS = 100;
static const uint32_t TIMEOUT_ATEM_MS = 2500;
static const uint32_t RELANCE_HELLO_MS = 1000;
static const uint8_t  TTL_MAX = 6;
static const int      LED_STATUT = 2;
static const uint8_t  NB_INTER = 4;
static const int      INTER_ME[NB_INTER] = {4, 14, 32, 33};    // IO4, IO14, IO32, IO33

// ---------- WT32-ETH01 : PHY LAN8720 ----------
static const int ETH_ADRESSE = 1;
static const int ETH_ALIM = 16;      // active l'oscillateur 50 MHz du PHY
static const int ETH_MDC_BROCHE = 23;
static const int ETH_MDIO_BROCHE = 18;

// ---------- Protocole ATEM ----------
static const uint8_t F_ACK_DEMANDE = 0x08;
static const uint8_t F_HELLO = 0x10;
static const uint8_t F_ACK = 0x80;
enum EtatLien { DECONNECTE, HELLO_ENVOYE, CONNECTE };
static const uint8_t BROADCAST[6] = {0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF};

// ---------- État de l'ATEM ----------
static const uint8_t MAX_ME = 4, MAX_CLES = 4, MAX_DSK = 4, MAX_SS = 2, MAX_BOITES = 4, MAX_MV = 4, MAX_FEN = 16;
struct Atem {
  uint16_t verMaj, verMin;
  uint8_t nbMe;
  uint16_t prg[MAX_ME], prv[MAX_ME];
  bool enTransition[MAX_ME];
  uint8_t selection[MAX_ME];                 // bit 0 fond, bits 1.. clés de la prochaine transition
  bool cleAir[MAX_ME][MAX_CLES];
  uint8_t cleType[MAX_ME][MAX_CLES];         // 3 = DVE
  uint16_t cleFill[MAX_ME][MAX_CLES];
  bool dskAir[MAX_DSK], dskTr[MAX_DSK], dskLie[MAX_DSK];
  uint16_t dskFill[MAX_DSK];
  bool ssActive[MAX_SS][MAX_BOITES];
  uint16_t ssSource[MAX_SS][MAX_BOITES], ssFond[MAX_SS];
  uint16_t mv[MAX_MV][MAX_FEN];
} atem;

EtatLien etat = DECONNECTE;
uint16_t sessionId = 0x53AB;
uint32_t dernierRecu = 0, dernierHello = 0, dernierRadio = 0, dernierLien = 0, dernierInter = 0;
uint8_t seq = 0;
uint8_t tally[40] = {0};
uint64_t pgmMe[MAX_ME], pvwMe[MAX_ME];       // bit k = entrée k+1
uint8_t masqueMe = 0xFF;
bool aRecalculer = false;
uint8_t rx[1600];
uint8_t tx[32];
int ledActuelle = -1;
WiFiUDP udp;
bool cableBranche = false;

void diffuser();

// ---------------------------------------------------------------------------
//  Calcul : quelles entrées sont visibles dans une source
// ---------------------------------------------------------------------------
uint64_t programme(int me, int prof, bool avecDsk = true);
uint64_t preview(int me, int prof);

uint64_t visible(uint16_t src, int prof) {
  if (prof > 6 || src == 0) return 0;                    // garde-fou contre les boucles
  if (src <= 40) return 1ULL << (src - 1);               // entrée physique
  if (src < 1000) return 0;
  if (src >= 6000 && src < 6000 + MAX_SS) {              // SuperSource
    int n = src - 6000;
    uint64_t m = visible(atem.ssFond[n], prof + 1);
    for (int b = 0; b < MAX_BOITES; b++) if (atem.ssActive[n][b]) m |= visible(atem.ssSource[n][b], prof + 1);
    return m;
  }
  if (src >= 7001 && src <= 7009) return programme(0, prof + 1, false);   // clean feed
  if (src >= 9001 && src < 9001 + MAX_MV) {              // multiview
    int n = src - 9001;
    uint64_t m = 0;
    for (int f = 0; f < MAX_FEN; f++) m |= visible(atem.mv[n][f], prof + 1);
    return m;
  }
  if (src >= 10010 && src <= 10999) {                    // sortie de M/E : 10010 prog M/E 1, 10011 prev
    int me = (src - 10000) / 10 - 1, quoi = src % 10;
    if (me < 0 || me >= MAX_ME) return 0;
    if (quoi == 0) return programme(me, prof + 1);
    if (quoi == 1) return preview(me, prof + 1);
  }
  return 0;
}

uint64_t programme(int me, int prof, bool avecDsk) {
  bool tr = atem.enTransition[me];
  uint8_t sel = atem.selection[me];
  uint64_t m = visible(atem.prg[me], prof);
  if (tr && (sel & 1)) m |= visible(atem.prv[me], prof);          // mix en cours
  for (int k = 0; k < MAX_CLES; k++)
    if (atem.cleAir[me][k] || (tr && (sel & (2 << k)))) m |= visible(atem.cleFill[me][k], prof);
  if (me == 0 && avecDsk)
    for (int d = 0; d < MAX_DSK; d++)
      if (atem.dskAir[d] || atem.dskTr[d]) m |= visible(atem.dskFill[d], prof);
  return m;
}

uint64_t preview(int me, int prof) {
  uint8_t sel = atem.selection[me];
  uint64_t m = visible((sel & 1) ? atem.prv[me] : atem.prg[me], prof);
  for (int k = 0; k < MAX_CLES; k++)
    if (atem.cleAir[me][k] != (bool)(sel & (2 << k))) m |= visible(atem.cleFill[me][k], prof);
  if (me == 0)
    for (int d = 0; d < MAX_DSK; d++)
      if (atem.dskAir[d] != atem.dskLie[d]) m |= visible(atem.dskFill[d], prof);
  return m;
}

int nbMeConnues() {
  int n = atem.nbMe;
  for (int m = 0; m < MAX_ME; m++) if (atem.prg[m] && n < m + 1) n = m + 1;
  return n < MAX_ME ? n : MAX_ME;
}

void listerEntrees(uint64_t m) {
  bool rien = true;
  for (int k = 0; k < NB_ENTREES; k++) if (m & (1ULL << k)) { Serial.printf("%s%d", rien ? "" : ",", k + 1); rien = false; }
  if (rien) Serial.print("-");
}

// Recalcule chaque M/E, puis combine les M/E choisies par les interrupteurs
void recalculer(bool forcer) {
  int nb = nbMeConnues();
  bool meChange = false;
  for (int m = 0; m < MAX_ME; m++) {
    uint64_t pg = m < nb ? programme(m, 0) : 0, pv = m < nb ? preview(m, 0) : 0;
    if (pg != pgmMe[m] || pv != pvwMe[m]) meChange = true;
    pgmMe[m] = pg; pvwMe[m] = pv;
  }
  if (meChange) {
    for (int m = 0; m < nb; m++) {
      Serial.printf("%sM/E %d PGM ", m ? " | " : "", m + 1); listerEntrees(pgmMe[m]);
      Serial.print(" PVW "); listerEntrees(pvwMe[m]);
    }
    Serial.println();
  }
  uint64_t pg = 0, pv = 0;
  for (int m = 0; m < nb; m++) if (masqueMe & (1 << m)) { pg |= pgmMe[m]; pv |= pvwMe[m]; }
  uint8_t n[40] = {0};
  for (int k = 0; k < NB_ENTREES; k++) n[k] = (pg & (1ULL << k)) ? 1 : (pv & (1ULL << k)) ? 2 : 0;
  if (!forcer && memcmp(n, tally, sizeof(tally)) == 0) return;
  memcpy(tally, n, sizeof(tally));
  Serial.print("Tally :");
  bool rien = true;
  for (int k = 0; k < NB_ENTREES; k++) if (tally[k]) { Serial.printf(" %d:%s", k + 1, tally[k] == 1 ? "PGM" : "PVW"); rien = false; }
  Serial.println(rien ? " aucune entrée" : "");
  diffuser();                                              // tout de suite, sans attendre
}

// ---------------------------------------------------------------------------
//  Lecture des commandes de l'ATEM
// ---------------------------------------------------------------------------
static inline uint16_t u16(const uint8_t *p) { return (p[0] << 8) | p[1]; }

void lireCommande(const char *nom, const uint8_t *c, int l) {
  bool v8 = atem.verMaj > 2 || (atem.verMaj == 2 && atem.verMin >= 28);
  if (!memcmp(nom, "_ver", 4) && l >= 4) { atem.verMaj = u16(c); atem.verMin = u16(c + 2); return; }
  if (!memcmp(nom, "_top", 4) && l >= 1) { atem.nbMe = c[0]; }
  else if (!memcmp(nom, "PrgI", 4) && l >= 4 && c[0] < MAX_ME) atem.prg[c[0]] = u16(c + 2);
  else if (!memcmp(nom, "PrvI", 4) && l >= 4 && c[0] < MAX_ME) atem.prv[c[0]] = u16(c + 2);
  else if (!memcmp(nom, "KeOn", 4) && l >= 3 && c[0] < MAX_ME && c[1] < MAX_CLES) atem.cleAir[c[0]][c[1]] = c[2];
  else if (!memcmp(nom, "KeBS", 4) && l >= 10 && c[0] < MAX_ME && c[1] < MAX_CLES) {
    atem.cleType[c[0]][c[1]] = c[2];
    atem.cleFill[c[0]][c[1]] = u16(c + 8);
  }
  else if (!memcmp(nom, "TrPs", 4) && l >= 2 && c[0] < MAX_ME) atem.enTransition[c[0]] = c[1] & 1;
  else if (!memcmp(nom, "TrSS", 4) && l >= 3 && c[0] < MAX_ME) atem.selection[c[0]] = c[2];
  else if (!memcmp(nom, "DskB", 4) && l >= 4 && c[0] < MAX_DSK) atem.dskFill[c[0]] = u16(c + 2);
  else if (!memcmp(nom, "DskS", 4) && l >= 3 && c[0] < MAX_DSK) { atem.dskAir[c[0]] = c[1]; atem.dskTr[c[0]] = c[2]; }
  else if (!memcmp(nom, "DskP", 4) && l >= 2 && c[0] < MAX_DSK) atem.dskLie[c[0]] = c[1];
  else if (!memcmp(nom, "SSBP", 4)) {
    if (v8 && l >= 6 && c[0] < MAX_SS && c[1] < MAX_BOITES) { atem.ssActive[c[0]][c[1]] = c[2]; atem.ssSource[c[0]][c[1]] = u16(c + 4); }
    else if (!v8 && l >= 4 && c[0] < MAX_BOITES) { atem.ssActive[0][c[0]] = c[1]; atem.ssSource[0][c[0]] = u16(c + 2); }
    else return;
  }
  else if (!memcmp(nom, "SSrc", 4)) {
    if (v8 && l >= 4 && c[0] < MAX_SS) atem.ssFond[c[0]] = u16(c + 2);
    else if (!v8 && l >= 2) atem.ssFond[0] = u16(c);
    else return;
  }
  else if (!memcmp(nom, "MvIn", 4) && l >= 4 && c[0] < MAX_MV && c[1] < MAX_FEN) atem.mv[c[0]][c[1]] = u16(c + 2);
  else if (!memcmp(nom, "InCm", 4)) { Serial.printf("État de l'ATEM reçu : %d M/E\n", atem.nbMe); }
  else return;
  aRecalculer = true;
}

// ---------------------------------------------------------------------------
//  Client ATEM
// ---------------------------------------------------------------------------
void envoyerUdp(const uint8_t *buf, uint16_t len) {
  udp.beginPacket(IPAddress(IP_ATEM[0], IP_ATEM[1], IP_ATEM[2], IP_ATEM[3]), PORT_ATEM);
  udp.write(buf, len);
  udp.endPacket();
}

void envoyerHello() {
  memset(tx, 0, 20);
  sessionId = 0x5300 | (esp_random() & 0xFF);
  tx[0] = F_HELLO; tx[1] = 20;
  tx[2] = sessionId >> 8; tx[3] = sessionId & 0xFF;
  tx[9] = 0x3A;
  tx[12] = 0x01;
  envoyerUdp(tx, 20);
  etat = HELLO_ENVOYE;
  dernierHello = millis();
}

void envoyerAck(uint16_t session, uint16_t id) {
  memset(tx, 0, 12);
  tx[0] = F_ACK; tx[1] = 12;
  tx[2] = session >> 8; tx[3] = session & 0xFF;
  tx[4] = id >> 8; tx[5] = id & 0xFF;
  envoyerUdp(tx, 12);
}

void oublierAtem() {
  memset(&atem, 0, sizeof(atem));
  for (int m = 0; m < MAX_ME; m++) atem.selection[m] = 1;   // par défaut : la transition prend le fond
  memset(pgmMe, 0, sizeof(pgmMe));
  memset(pvwMe, 0, sizeof(pvwMe));
  memset(tally, 0, sizeof(tally));
}

void traiterPaquet(int n) {
  if (n < 12) return;
  uint8_t drapeaux = rx[0] & 0xF8;
  int longueur = ((rx[0] & 0x07) << 8) | rx[1];
  if (longueur < 12 || longueur > n) return;
  uint16_t session = (rx[2] << 8) | rx[3];
  uint16_t idDistant = (rx[10] << 8) | rx[11];

  if (drapeaux & F_HELLO) {
    if (longueur >= 13 && rx[12] == 0x02) {
      envoyerAck(session, 0);
      oublierAtem();
      etat = CONNECTE;
      dernierRecu = millis();
      Serial.println("Connecté à l'ATEM, lecture de son état...");
    } else if (longueur >= 13 && rx[12] == 0x03) {
      Serial.println("L'ATEM refuse la connexion (trop de clients ?)");
    }
    return;
  }
  if (etat != CONNECTE) return;
  dernierRecu = millis();
  sessionId = session;
  if (drapeaux & F_ACK_DEMANDE) envoyerAck(session, idDistant);

  int i = 12;
  while (i + 8 <= longueur) {
    int l = (rx[i] << 8) | rx[i + 1];
    if (l < 8 || i + l > longueur) break;
    lireCommande((const char *)&rx[i + 4], &rx[i + 8], l - 8);
    i += l;
  }
}

// ---------------------------------------------------------------------------
//  Interrupteurs et câble réseau
// ---------------------------------------------------------------------------
uint8_t lireInterrupteurs() {
  uint8_t m = 0;
  for (int i = 0; i < NB_INTER; i++) if (digitalRead(INTER_ME[i]) == LOW) m |= 1 << i;
  return m ? m : 0xFF;
}

void afficherChoix() {
  if (masqueMe == 0xFF) { Serial.println("M/E suivies : toutes"); return; }
  Serial.print("M/E suivies :");
  for (int i = 0; i < NB_INTER; i++) if (masqueMe & (1 << i)) Serial.printf(" %d", i + 1);
  Serial.println();
}

void surveillerCable() {
  bool b = ETH.linkUp();
  if (b == cableBranche) return;
  cableBranche = b;
  etat = DECONNECTE;
  dernierHello = 0;
  oublierAtem();
  Serial.println(b ? "Câble réseau branché" : "Câble réseau débranché !");
}

// ---------------------------------------------------------------------------
//  ESP-NOW : trame v2, 8 octets d'en-tête + une case par entrée
// ---------------------------------------------------------------------------
void diffuser() {
  bool ok = etat == CONNECTE;
  uint8_t m[8 + 40];
  m[0] = 'T'; m[1] = 'L'; m[2] = 2;
  m[3] = seq++;
  m[4] = ok ? 1 : 0;
  m[5] = NB_ENTREES;
  m[6] = TTL_MAX;
  m[7] = 0;
  for (int k = 0; k < NB_ENTREES; k++) m[8 + k] = ok ? tally[k] : 0;
  esp_now_send(BROADCAST, m, 8 + NB_ENTREES);
  dernierRadio = millis();
}

void demarrerEspNow() {
  WiFi.mode(WIFI_STA);
  WiFi.disconnect();
  esp_wifi_set_promiscuous(true);
  esp_wifi_set_channel(CANAL_ESPNOW, WIFI_SECOND_CHAN_NONE);
  esp_wifi_set_promiscuous(false);
  if (esp_now_init() != ESP_OK) { Serial.println("Erreur esp_now_init"); return; }
  esp_now_peer_info_t pair = {};
  memcpy(pair.peer_addr, BROADCAST, 6);
  pair.channel = CANAL_ESPNOW;
  pair.ifidx = WIFI_IF_STA;
  pair.encrypt = false;
  if (esp_now_add_peer(&pair) != ESP_OK) Serial.println("Erreur esp_now_add_peer");
}

void setup() {
  Serial.begin(115200);
  pinMode(LED_STATUT, OUTPUT);
  digitalWrite(LED_STATUT, HIGH);
  for (int i = 0; i < NB_INTER; i++) pinMode(INTER_ME[i], INPUT_PULLUP);
  delay(5);
  masqueMe = lireInterrupteurs();
  oublierAtem();
  demarrerEspNow();

#if ESP_ARDUINO_VERSION_MAJOR >= 3
  ETH.begin(ETH_PHY_LAN8720, ETH_ADRESSE, ETH_MDC_BROCHE, ETH_MDIO_BROCHE, ETH_ALIM, ETH_CLOCK_GPIO0_IN);
#else
  ETH.begin(ETH_ADRESSE, ETH_ALIM, ETH_MDC_BROCHE, ETH_MDIO_BROCHE, ETH_PHY_LAN8720, ETH_CLOCK_GPIO0_IN);
#endif
  IPAddress ip(IP_EMETTEUR[0], IP_EMETTEUR[1], IP_EMETTEUR[2], IP_EMETTEUR[3]);
  IPAddress masque(MASQUE[0], MASQUE[1], MASQUE[2], MASQUE[3]);
  IPAddress routeur(IP_EMETTEUR[0], IP_EMETTEUR[1], IP_EMETTEUR[2], 1);
  ETH.config(ip, routeur, masque);               // IP fixe (pas de DHCP)
  udp.begin(PORT_LOCAL);
  Serial.printf("Ethernet prêt, WT32 en %d.%d.%d.%d\n", IP_EMETTEUR[0], IP_EMETTEUR[1], IP_EMETTEUR[2], IP_EMETTEUR[3]);
  surveillerCable();
  Serial.printf("Émetteur prêt, ATEM %d.%d.%d.%d, ESP-NOW canal %d, entrées 1 à %d\n",
                IP_ATEM[0], IP_ATEM[1], IP_ATEM[2], IP_ATEM[3], CANAL_ESPNOW, NB_ENTREES);
  afficherChoix();
}

void loop() {
  if (millis() - dernierLien >= 1000) { surveillerCable(); dernierLien = millis(); }

  // Paquets de l'ATEM (au plus 20 par tour pour garder la radio régulière)
  for (int k = 0; k < 20; k++) {
    int n = udp.parsePacket();
    if (n <= 0) break;
    int lu = udp.read(rx, sizeof(rx));
    IPAddress ip = udp.remoteIP();
    if (ip[0] == IP_ATEM[0] && ip[1] == IP_ATEM[1] && ip[2] == IP_ATEM[2] && ip[3] == IP_ATEM[3]) traiterPaquet(lu);
  }
  if (aRecalculer) { aRecalculer = false; recalculer(false); }

  if (millis() - dernierInter >= 50) {
    dernierInter = millis();
    uint8_t m = lireInterrupteurs();
    if (m != masqueMe) { masqueMe = m; afficherChoix(); recalculer(true); }
  }

  if (cableBranche && etat != CONNECTE && millis() - dernierHello >= RELANCE_HELLO_MS) envoyerHello();
  if (etat == CONNECTE && millis() - dernierRecu > TIMEOUT_ATEM_MS) {
    etat = DECONNECTE;
    oublierAtem();
    Serial.println("ATEM perdu, nouvelle tentative...");
  }

  int led = (etat == CONNECTE) ? LOW : HIGH;
  if (led != ledActuelle) { digitalWrite(LED_STATUT, led); ledActuelle = led; }
  if (millis() - dernierRadio >= PERIODE_RADIO_MS) diffuser();
}

Tally · tally_v2.ino

Carte : ESP32C3 Dev Module, « USB CDC On Boot : Enabled ». Bibliothèque « Adafruit NeoPixel ». Rien à modifier : chaque tally démarre sur l'entrée 1, puis on la règle avec les boutons + et −.

// ============================================================================
//  VERSION 2 · Tally  ·  ESP32-C3 + LED WS2812B + afficheur 7 segments 2 chiffres
// ----------------------------------------------------------------------------
//  Le même code sur tous les tallys : l'entrée SDI lue se règle avec les
//  boutons + et −, et s'affiche sur l'afficheur 7 segments double (« 7 », « 12 »).
//
//  Câblage (ESP32-C3 Super Mini) :
//    GPIO4  -> 330 Ω -> DIN de la WS2812B      5V  -> VDD,  GND -> GND
//    Deux chiffres 7 segments séparés (5161AS, cathode commune), un 74HC595 chacun :
//    GPIO5  -> DS (broche 14) du 1er 74HC595   GPIO6 -> SH_CP (11) des deux 74HC595
//    GPIO3  -> ST_CP (12) des deux 74HC595     Q7' (9) du 1er -> DS (14) du 2e
//    1er 74HC595 : unités, 2e : dizaines. Sorties Q0..Q6 -> 220 Ω -> segments a..g
//    VCC (16) et MR (10) -> 3V3, GND (8) et OE (13) -> GND, cathodes communes -> GND
//    GPIO7  -> bouton « + » -> GND             GPIO10 -> bouton « − » -> GND
//
//  Boutons :
//    appui court     -> entrée suivante (+) ou précédente (−), de 1 à 20
//    appui maintenu  -> défile tout seul
//    L'afficheur clignote tant que le choix n'est pas enregistré ; il est gardé
//    en mémoire 3 s après le dernier appui, même après une coupure.
//
//  LED, une seule couleur à la fois, priorité BLEU > ROUGE > VERT :
//    bleu fixe        -> plus aucune trame reçue depuis 2 s
//    bleu clignotant  -> l'émetteur répond mais n'a plus l'ATEM
//    rouge            -> entrée au programme
//    vert             -> entrée en preview
//    éteinte          -> hors antenne
//
//  Maillage : chaque trame nouvelle est rediffusée une fois (TTL - 1).
//  Bibliothèque : « Adafruit NeoPixel ».
//  Carte : « ESP32C3 Dev Module », USB CDC On Boot : Enabled.
// ============================================================================
#include <WiFi.h>
#include <esp_now.h>
#include <esp_wifi.h>
#include <Preferences.h>
#include <Adafruit_NeoPixel.h>

static const uint8_t  ENTREE_DEPART = 1;     // entrée utilisée la toute première fois
static const uint8_t  MAX_ENTREES = 20;     // entrées envoyées par l'émetteur
static const bool     RELAIS = true;
static const uint8_t  CANAL_ESPNOW = 1;
static const uint32_t TIMEOUT_MS = 2000;
static const int      PIN_WS2812 = 4;        // GPIO4
static const uint8_t  LUMINOSITE_LED = 160;  // 0 à 255
static const int      SR_DATA = 5;           // GPIO5 -> DS du 1er 74HC595
static const int      SR_HORLOGE = 6;        // GPIO6 -> SH_CP des deux 74HC595
static const int      SR_VERROU = 3;         // GPIO3 -> ST_CP des deux 74HC595
static const bool     ANODE_COMMUNE = false; // true si l'afficheur est à anode commune
static const int      BOUTON_PLUS = 7;       // GPIO7
static const int      BOUTON_MOINS = 10;     // GPIO10

static const uint8_t  TRAME_MIN = 16;
static const uint8_t  TRAME_MAX = 8 + 40;
static const uint8_t  BROADCAST[6] = {0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF};

portMUX_TYPE verrou = portMUX_INITIALIZER_UNLOCKED;
volatile uint8_t entree = ENTREE_DEPART;     // lue aussi dans le callback radio
volatile uint32_t dernierMessage = 0;
volatile bool dejaRecu = false;
volatile uint8_t dernierSeq = 0;
volatile uint8_t atemOk = 0;
volatile uint8_t etatCam = 0;
volatile uint8_t sauts = 0;
uint8_t relais[TRAME_MAX];
volatile uint8_t relaisLen = 0;
volatile bool relaisEnAttente = false;
volatile uint32_t relaisA = 0;

Preferences memoire;
Adafruit_NeoPixel led(1, PIN_WS2812, NEO_GRB + NEO_KHZ800);
uint32_t couleurActuelle = 0xFFFFFFFF;
int etatAvant = -1;
uint8_t sautsAffiches = 0xFF;

// ---------------------------------------------------------------------------
//  Afficheur 7 segments 2 chiffres par deux 74HC595 (sans bibliothèque)
// ---------------------------------------------------------------------------
//  Bits : Q0 = a, Q1 = b, Q2 = c, Q3 = d, Q4 = e, Q5 = f, Q6 = g, Q7 = point.
static const uint8_t CHIFFRES[10] = {0x3F, 0x06, 0x5B, 0x4F, 0x66, 0x6D, 0x7D, 0x07, 0x7F, 0x6F};

void envoyerOctet(uint8_t b) {
  if (ANODE_COMMUNE) b = ~b;
  for (int i = 7; i >= 0; i--) {             // Q7 d'abord, Q0 en dernier
    digitalWrite(SR_DATA, (b >> i) & 1);
    digitalWrite(SR_HORLOGE, HIGH);
    digitalWrite(SR_HORLOGE, LOW);
  }
}

int segAffiche = -1;
void afficherEntree(uint8_t n, bool allume) {
  int code = n | (allume ? 0x100 : 0);
  if (code == segAffiche) return;
  segAffiche = code;
  uint8_t dizaines = (allume && n >= 10) ? CHIFFRES[(n / 10) % 10] : 0;
  uint8_t unites = allume ? CHIFFRES[n % 10] : 0;
  digitalWrite(SR_VERROU, LOW);
  envoyerOctet(dizaines);                    // part au bout de la chaîne : 2e 74HC595
  envoyerOctet(unites);                      // reste dans le 1er 74HC595
  digitalWrite(SR_VERROU, HIGH);             // les deux chiffres changent en même temps
}

// ---------------------------------------------------------------------------
//  Radio
// ---------------------------------------------------------------------------
static bool estNouvelle(uint8_t seq, uint32_t maintenant) {
  if (!dejaRecu || maintenant - dernierMessage > 1000) return true;
  uint8_t d = seq - dernierSeq;
  return d > 0 && d < 128;
}

static void traiterTrame(const uint8_t *data, int len) {
  if (len < TRAME_MIN || data[0] != 'T' || data[1] != 'L' || data[2] != 2) return;
  uint32_t maintenant = millis();
  portENTER_CRITICAL(&verrou);
  if (!estNouvelle(data[3], maintenant)) { portEXIT_CRITICAL(&verrou); return; }
  dernierSeq = data[3];
  dernierMessage = maintenant;
  dejaRecu = true;
  atemOk = data[4];
  uint8_t e = entree;
  etatCam = (e <= data[5] && 7 + e < len) ? data[7 + e] : 0;
  sauts = data[7] + 1;
  if (RELAIS && data[6] > 1) {
    relaisLen = len < TRAME_MAX ? len : TRAME_MAX;
    memcpy(relais, data, relaisLen);
    relais[6] = data[6] - 1;
    relais[7] = data[7] + 1;
    relaisA = maintenant + 2 + (esp_random() % 13);
    relaisEnAttente = true;
  }
  portEXIT_CRITICAL(&verrou);
}

#if ESP_ARDUINO_VERSION_MAJOR >= 3
void surReception(const esp_now_recv_info_t *info, const uint8_t *data, int len) { traiterTrame(data, len); }
#else
void surReception(const uint8_t *mac, const uint8_t *data, int len) { traiterTrame(data, len); }
#endif

void afficherCouleur(uint32_t c) {
  if (c == couleurActuelle) return;           // la WS2812B garde sa couleur
  led.setPixelColor(0, c);
  led.show();
  couleurActuelle = c;
}

// ---------------------------------------------------------------------------
//  Boutons + et − : appui court = un pas, maintenu = défilement ; enregistré 3 s après
// ---------------------------------------------------------------------------
struct Bouton { int broche; int sens; bool avant; uint32_t debut, dernierPas; };
Bouton boutons[2] = {{BOUTON_PLUS, +1, false, 0, 0}, {BOUTON_MOINS, -1, false, 0, 0}};
uint32_t dernierChangement = 0;
bool aEnregistrer = false;

void changerEntree(int sens) {
  int n = entree + sens;
  if (n > MAX_ENTREES) n = 1;
  if (n < 1) n = MAX_ENTREES;
  portENTER_CRITICAL(&verrou);
  entree = n;
  etatCam = 0;                               // attendre la prochaine trame pour la nouvelle entrée
  portEXIT_CRITICAL(&verrou);
  dernierChangement = millis();
  aEnregistrer = true;
  Serial.printf("Entrée SDI : %d\n", n);
}

void gererBoutons() {
  uint32_t t = millis();
  bool unAppui = false;
  for (auto &b : boutons) {
    bool appui = digitalRead(b.broche) == LOW;
    if (appui && !b.avant && t - b.debut > 30) { b.debut = t; b.dernierPas = t; changerEntree(b.sens); }
    if (appui && b.avant && t - b.debut > 700 && t - b.dernierPas > 200) { b.dernierPas = t; changerEntree(b.sens); }
    b.avant = appui;
    unAppui |= appui;
  }
  if (aEnregistrer && !unAppui && t - dernierChangement > 3000) {
    memoire.putUChar("entree", entree);
    aEnregistrer = false;
    Serial.printf("Entrée %d enregistrée\n", entree);
  }
}

void setup() {
  Serial.begin(115200);
  led.begin();
  led.setBrightness(LUMINOSITE_LED);
  pinMode(BOUTON_PLUS, INPUT_PULLUP);
  pinMode(BOUTON_MOINS, INPUT_PULLUP);
  pinMode(SR_DATA, OUTPUT);
  pinMode(SR_HORLOGE, OUTPUT);
  pinMode(SR_VERROU, OUTPUT);

  memoire.begin("tally", false);
  uint8_t e = memoire.getUChar("entree", ENTREE_DEPART);
  entree = (e >= 1 && e <= MAX_ENTREES) ? e : ENTREE_DEPART;
  afficherEntree(entree, true);

  // Petit test au démarrage : rouge, vert, bleu, 300 ms chacun
  afficherCouleur(led.Color(255, 0, 0)); delay(300);
  afficherCouleur(led.Color(0, 255, 0)); delay(300);
  afficherCouleur(led.Color(0, 0, 255)); delay(300);
  afficherCouleur(0);

  WiFi.mode(WIFI_STA);
  WiFi.disconnect();
  esp_wifi_set_promiscuous(true);
  esp_wifi_set_channel(CANAL_ESPNOW, WIFI_SECOND_CHAN_NONE);
  esp_wifi_set_promiscuous(false);
  if (esp_now_init() != ESP_OK) { Serial.println("Erreur esp_now_init"); return; }
  esp_now_register_recv_cb(surReception);
  esp_now_peer_info_t pair = {};
  memcpy(pair.peer_addr, BROADCAST, 6);
  pair.channel = CANAL_ESPNOW;
  pair.ifidx = WIFI_IF_STA;
  pair.encrypt = false;
  if (esp_now_add_peer(&pair) != ESP_OK) Serial.println("Erreur esp_now_add_peer");
  Serial.printf("Tally prêt sur l'entrée SDI %d, canal %d, relais %s\n", entree, CANAL_ESPNOW, RELAIS ? "actif" : "inactif");
}

void loop() {
  uint8_t aEnvoyer[TRAME_MAX];
  uint8_t nEnvoi = 0;
  portENTER_CRITICAL(&verrou);
  if (relaisEnAttente && (int32_t)(millis() - relaisA) >= 0) {
    nEnvoi = relaisLen;
    memcpy(aEnvoyer, relais, nEnvoi);
    relaisEnAttente = false;
  }
  uint32_t dernier = dernierMessage;
  bool recu = dejaRecu;
  uint8_t ok = atemOk, e = etatCam, s = sauts;
  portEXIT_CRITICAL(&verrou);
  if (nEnvoi) esp_now_send(BROADCAST, aEnvoyer, nEnvoi);

  gererBoutons();
  uint32_t maintenant = millis();
  afficherEntree(entree, !aEnregistrer || (maintenant % 400) < 250);

  // 0 : plus de trame, 1 : ATEM perdu, 2 : programme, 3 : preview, 4 : rien
  int etat = 4;
  if (!recu || maintenant - dernier > TIMEOUT_MS) etat = 0;
  else if (ok == 0) etat = 1;
  else if (e & 0x01) etat = 2;
  else if (e & 0x02) etat = 3;

  if (etat != etatAvant) {
    static const char *noms[] = {"plus de trame tally : bleu", "émetteur sans ATEM : bleu clignotant",
                                 "programme : rouge", "preview : vert", "hors antenne : éteint"};
    Serial.printf("Entrée %d : %s\n", entree, noms[etat]);
    etatAvant = etat;
  }
  if (etat != 0 && s != sautsAffiches) {
    Serial.printf("Trame reçue en %d saut(s)\n", s);
    sautsAffiches = s;
  }

  const uint32_t BLEU = led.Color(0, 0, 255), ROUGE = led.Color(255, 0, 0), VERT = led.Color(0, 255, 0);
  switch (etat) {
    case 0: afficherCouleur(BLEU); break;
    case 1: afficherCouleur((maintenant % 1000) < 500 ? BLEU : 0); break;
    case 2: afficherCouleur(ROUGE); break;
    case 3: afficherCouleur(VERT); break;
    default: afficherCouleur(0);
  }
  delay(1);
}
